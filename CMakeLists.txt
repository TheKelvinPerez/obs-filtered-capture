cmake_minimum_required(VERSION 3.28)
project(obs-filtered-capture VERSION 0.1.0 LANGUAGES C OBJC)
if(NOT APPLE)
  message(FATAL_ERROR "This plugin requires macOS")
endif()
set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0")
set(OBS_SOURCE_DIR "" CACHE PATH "OBS Studio 33.0.1 source checkout")
set(OBS_APP "/Applications/OBS.app" CACHE PATH "Installed OBS Studio")
set(SIMDE_INCLUDE_DIR "" CACHE PATH "Directory containing simde/x86/sse2.h")
if(NOT EXISTS "${OBS_SOURCE_DIR}/libobs/obs-module.h")
  message(FATAL_ERROR "Set OBS_SOURCE_DIR to an OBS 33.0.1 source checkout")
endif()
set(OBS_LIBRARY "${OBS_APP}/Contents/Frameworks/libobs.framework/Versions/A/libobs")
if(NOT EXISTS "${OBS_LIBRARY}")
  message(FATAL_ERROR "Cannot find installed libobs at ${OBS_LIBRARY}")
endif()
set(OBS_RELEASE_CANDIDATE 0)
set(OBS_BETA 0)
configure_file("${OBS_SOURCE_DIR}/libobs/obsconfig.h.in" "${CMAKE_CURRENT_BINARY_DIR}/generated/obsconfig.h")
add_library(obs-filtered-capture MODULE src/filtered-capture.m)
set_target_properties(obs-filtered-capture PROPERTIES PREFIX "" SUFFIX "" BUNDLE TRUE BUNDLE_EXTENSION "plugin" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in" BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@executable_path/../Frameworks")
target_include_directories(obs-filtered-capture PRIVATE "${OBS_SOURCE_DIR}/libobs" "${SIMDE_INCLUDE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_options(obs-filtered-capture PRIVATE -fobjc-arc -fvisibility=hidden -Wall -Wextra -Wno-unused-parameter)
target_link_libraries(obs-filtered-capture PRIVATE "${OBS_LIBRARY}" "-framework AppKit" "-framework ScreenCaptureKit" "-framework CoreMedia" "-framework CoreVideo" "-framework IOSurface")
enable_testing()
add_executable(policy-tests tests/policy-tests.m)
target_include_directories(policy-tests PRIVATE src)
target_compile_options(policy-tests PRIVATE -fobjc-arc -UNDEBUG)
target_link_libraries(policy-tests PRIVATE "-framework Foundation")
add_test(NAME exclusion-policy COMMAND policy-tests)
add_executable(module-smoke tests/module-smoke.m)
target_include_directories(module-smoke PRIVATE "${OBS_SOURCE_DIR}/libobs" "${SIMDE_INCLUDE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_options(module-smoke PRIVATE -fobjc-arc -UNDEBUG)
target_compile_definitions(module-smoke PRIVATE FC_OBS_APP_PATH="${OBS_APP}")
target_link_libraries(module-smoke PRIVATE "${OBS_LIBRARY}" "-framework AppKit")
set_target_properties(module-smoke PROPERTIES BUILD_RPATH "${OBS_APP}/Contents/Frameworks")
add_test(NAME module-smoke COMMAND module-smoke "$<TARGET_FILE:obs-filtered-capture>")
set_tests_properties(module-smoke PROPERTIES TIMEOUT 60)
add_test(NAME installer-safety COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/tests/installer-test.sh")
